──────────────────────────────────────────────────────────────────
1. Verify pods are running
──────────────────────────────────────────────────────────────────

  kubectl get pods -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}
  kubectl get statefulsets,deployments,services -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}

──────────────────────────────────────────────────────────────────
2. Tail the chain-node logs
──────────────────────────────────────────────────────────────────

{{- range $r := list "validator" "sentry" "full" "rpc" "archive" "bootnode" }}
{{- $cfg := index $.Values.chainNode $r | default dict }}
{{- if $cfg.enabled }}
  kubectl logs -n {{ $.Release.Namespace }} -f -l viper.pqchain.io/role={{ $r }} --max-log-requests=10
{{- end }}
{{- end }}

──────────────────────────────────────────────────────────────────
3. Hit the public read API
──────────────────────────────────────────────────────────────────

{{- $rpcSvc := include "viper-pq-chain.firstChainNodeService" . -}}
{{- if $rpcSvc }}

  # In-cluster:
  kubectl run -i --rm tmp-curl --image=curlimages/curl -- \
    curl -s "http://{{ $rpcSvc }}:26657/v1/status" | jq .

  # Or port-forward:
  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ $rpcSvc }} 26657:26657
  curl -s http://127.0.0.1:26657/v1/status | jq .
{{- else }}

  No chain-node role enabled — nothing to query. Set at least one of:
    chainNode.{validator|sentry|full|rpc|archive}.enabled = true
{{- end }}


──────────────────────────────────────────────────────────────────
4. Run the helm test
──────────────────────────────────────────────────────────────────

  helm test {{ .Release.Name }} -n {{ .Release.Namespace }}

──────────────────────────────────────────────────────────────────
5. Uninstall
──────────────────────────────────────────────────────────────────

  helm uninstall {{ .Release.Name }} -n {{ .Release.Namespace }}
  # PVCs are NOT deleted by default — clean up manually if you want
  # to drop chain state:
  kubectl delete pvc -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}
